	\subsection{Producer-Consumer with Unbounded Queue}
	(defparameter *work-queue* (make-queue))
	\begin{lstlisting}
;; Producer thread
(defun producer ()
(loop for i from 1 to 1000 do
(queue-push *work-queue* i)
(sleep 0.001)))

;; Consumer thread
(defun consumer ()
(loop
(multiple-value-bind (item found)
(queue-pop *work-queue*)
(if found
(format t "Processing: ~A~%" item)
(sleep 0.001)))))
	\end{lstlisting}
